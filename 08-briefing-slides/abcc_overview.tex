\section{The Competition at a Glance}
% ============================================================

\subsection{Host and Setting}

\begin{frame}{What is the ABCC?}
  \begin{block}{Asper Business Case Competition (ABCC)}
    A five-day, graduate-level business case competition hosted by the
    I.\,H.~Asper School of Business, University of Manitoba, in
    Winnipeg, Canada. The 2027 edition (February 15--20) is the
    \textbf{fifth annual} competition.
  \end{block}
  \begin{itemize}
    \item Hosted at the \textbf{Drake Centre}, 181 Freedman Crescent, University of Manitoba campus
    \item Teams from top Canadian business schools -- and, since 2026, international teams
    \item 2026 field: 9 graduate teams, incl.\ HEC Montr\'eal, DeGroote, Beedie, Babe\c{s}-Bolyai (Romania)
  \end{itemize}
\end{frame}

\begin{frame}{Why It Matters for Us}
  \begin{itemize}
    \item First-hand experience with the \textbf{North American case method} under time pressure
    \item Benchmarking against strong Canadian MBA/graduate teams -- the 2025 field included Ivey, John Molson, HEC Montr\'eal, Telfer, DeGroote, Beedie, Levene, Memorial and Asper
    \item In 2026, the international newcomer Babe\c{s}-Bolyai University \textbf{won} the competition -- international teams can absolutely compete for the title
    \item Rich social program: team-bonding events, networking with judges from industry, gala dinner with awards
  \end{itemize}
  \begin{exampleblock}{Scale of the event (2026)}
    35 participants, 9 faculty advisors, 50 industry judges, 52 volunteers.
  \end{exampleblock}
\end{frame}

\subsection{Team and Eligibility}

\begin{frame}{Team Composition and Eligibility}
  \begin{itemize}
    \item Teams consist of \textbf{four students}; all four present in every round
    \item Open to graduate-level students (full-time, part-time, executive MBA, and Masters-level students of other management programs)
    \item Each team travels with a \textbf{faculty advisor / coach} (that is me)
    \item Maximum of four laptops per team; \textbf{no cellphones or smartwatches} during case preparation and presentations
  \end{itemize}
  \begin{alertblock}{Golden rule}
    \textbf{Every team member must speak} in every presentation and discussion round; judging uses the official 7-category team rubric.
  \end{alertblock}
\end{frame}

\begin{frame}{Three Rounds, Three Different Styles}
  \begin{center}
    \begin{tabular}{llll}
      \toprule
      \textbf{Round} & \textbf{Prep time} & \textbf{Format} & \textbf{Weight} \\
      \midrule
      1: Prepared case    & ca.\ 2 weeks (before travel) & 25\,min + 15\,min Q\&A & 25\,\% \\
      2: Boardroom case   & short (4\,h in 2026)         & 3\,min + 22\,min discussion & 30\,\% \\
      3: 10-hour case & \textbf{10\,h} (2026: 3\,h)         & 20\,min + 10\,min Q\&A & 45\,\% \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Escalating difficulty and point value across the three rounds
    \item Top 3 teams advance to a \textbf{final round} in front of a new judge panel
    \item Ties broken by the highest third-round (traditional) score
  \end{itemize}
\end{frame}

\subsection{Round 1: Prepared Case}

\begin{frame}{Round 1: The Prepared Case}
  \begin{block}{Format}
    25-minute presentation + 15-minute Q\&A. Unlimited slides and exhibits. Time warnings at 5 minutes (yellow) and 1 minute (red).
  \end{block}
  \begin{itemize}
    \item Case is released \textbf{weeks before the competition} -- worked on at home
    \item A real client: in 2026 this was \textbf{Jazz Winnipeg} (pitch a radical new operational idea)
    \item All resources may be used -- \textbf{except} contacting employees of the case sponsor
    \item The coach \textbf{may} give input on this case (unlike rounds 2 and 3)
    \item Slide deck must be \textbf{submitted before the deadline} -- late submissions are placed at the bottom of the pool
  \end{itemize}
\end{frame}

\subsection{Round 2: Boardroom Case}

\begin{frame}{Round 2: The Boardroom Case}
  \begin{block}{Format}
    Seated boardroom setting: 3-minute uninterrupted recommendation, then 22 minutes of open discussion with the judges (in 2026: four judges plus a company representative).
  \end{block}
  \begin{itemize}
    \item Short preparation window on site (4 hours in 2025 and 2026)
    \item Protected 3 minutes: \textbf{one summary slide only} -- appendices are blocked; red card at 1 minute
    \item Appendices unlocked for the 22-minute discussion (yellow/red cards at 5/1 minutes)
    \item Judges receive a \textbf{one-page black-and-white brief} in advance; deck + brief handed in on a provided USB stick
    \item Seated delivery via wireless keyboard/mouse; \textbf{no coach contact} after case release
    \item 2026 client: Coal \& Canary -- elevate the brand from premium to luxury
  \end{itemize}
\end{frame}

\subsection{Round 3: The 10-Hour Case}

\begin{frame}{Round 3: Traditional Case $\rightarrow$ 10-Hour Case}
  \begin{block}{Format}
    Classic case-competition style: 20-minute presentation + 10-minute Q\&A. Time warnings at 5 minutes (yellow) and 1 minute (red).
  \end{block}
  \begin{itemize}
    \item 2027: listed as a \textbf{10-hour case} -- prep on Friday, presentation on Saturday (2025/2026: 3 hours, same day). Details to be confirmed
    \item Highest weight of all rounds: \textbf{45\,\%} of the total score -- and the deck the finalists re-present
    \item One laptop per student; no cellphones or smartwatches
    \item \textbf{No coach contact} after case release
    \item 2026 client: Access Credit Union -- service innovations to engage young customers
  \end{itemize}
\end{frame}

\subsection{Final Round}

\begin{frame}{The Final Round}
  \begin{itemize}
    \item The \textbf{top 3 teams} present again before a fresh judge panel
    \item 20-minute presentation + 15-minute Q\&A
    \item Slide decks \textbf{cannot be modified} for the final
    \item Faculty advisors may not communicate with their team until after the presentation
    \item Winners announced at the \textbf{awards dinner} (2027: Sat Feb 20)
  \end{itemize}
  \begin{exampleblock}{2026 podium}
    1.~Babe\c{s}-Bolyai University (Romania) \quad 2.~DeGroote School of Business \quad 3.~HEC Montr\'eal
  \end{exampleblock}
\end{frame}

\section{Rules and Judging}
% ============================================================

\subsection{Core Rules}

\begin{frame}{Core Rules Across All Rounds}
  \begin{itemize}
    \item Competition language: \textbf{English}; presentations must be \textbf{PowerPoint (.pptx)} -- no Prezi etc.
    \item All four team members must speak in every round (not necessarily during the protected 3 minutes of the boardroom pitch)
    \item One laptop per student; no cellphones or smartwatches during prep and presentation
    \item No contact with the coach after release of the boardroom and the third-round (10-hour) case; on-site decks are collected on a provided USB stick at fixed pickup times
    \item Time warnings: yellow card at 5 minutes, red card at 1 minute; \textbf{late arrival = no extra time}
    \item Slide clickers are provided; volunteers monitor rule compliance
  \end{itemize}
\end{frame}

\begin{frame}{Fair Play, Anonymity and Allowed Tools}
  \begin{alertblock}{Disqualification risks}
    Revealing your \textbf{school of origin} to a judge at any point, or any coach communication during a presentation or Q\&A, leads to \textbf{disqualification}. Teams compete under neutral letters (division draw at registration).
  \end{alertblock}
  \begin{itemize}
    \item \textbf{Internet use is allowed} during prep -- including university databases and \textbf{Generative AI}
    \item Prebuilt \textbf{own} slide templates are permitted (built outside prep time); no third-party templates for the prepared case
    \item Never contact employees of a case sponsor; cases are confidential -- no sharing or reproducing
    \item Dress code: \textbf{business professional} throughout competition days
  \end{itemize}
\end{frame}

\subsection{Scoring}

\begin{frame}{Scoring and Tie-Breaking}
  \begin{center}
    \begin{tabular}{lr}
      \toprule
      \textbf{Round} & \textbf{Weight} \\
      \midrule
      Prepared case    & 25\,\% \\
      Boardroom case   & 30\,\% \\
      Traditional / 10-hour case & 45\,\% \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Rubric-based evaluation by panels of industry judges
    \item Ties resolved by highest third-round score, then by performance in the weighted rounds and individual rubric categories
    \item Strategic implication: \textbf{the on-site rounds decide the competition} (75\,\% combined) -- 4\,h of speed and 10\,h of depth; our training covers both
  \end{itemize}
\end{frame}

\begin{frame}{The Judging Rubric (Official, 2025 Event Guide)}
  \begin{center}
    \begin{tabular}{lc}
      \toprule
      \textbf{Category} & \textbf{Weight} \\
      \midrule
      Implementation \& resources & $2.5\times$ \\
      Analysis & $2.0\times$ \\
      Recommendation & $2.0\times$ \\
      Communicating effectively & $1.0\times$ \\
      Framing the problem & $1.0\times$ \\
      Impact & $1.0\times$ \\
      Measurement, risks \& mitigation & $1.0\times$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{alertblock}{Training implication}
    \textbf{Implementation \& resources carries the highest weight} (2.5$\times$): every recommendation needs a concrete path -- steps, owners, timeline, resources, and financials -- not just a good idea.
  \end{alertblock}
\end{frame}

\subsection{Preparation Roadmap}

\begin{frame}{Preparation Roadmap (Details in the Prep Plan)}
  \begin{enumerate}
    \item \textbf{Foundations} -- case method, core frameworks, business math drills
    \item \textbf{Skill building} -- role-specific deep dives, slide craft, storylining
    \item \textbf{Simulation} -- timed mock cases: 4\,h boardroom, 3\,h speed runs, and full 10-hour days
    \item \textbf{Prepared case sprint} -- structured 2-week sprint once Case 1 is released
    \item \textbf{Final polish} -- Q\&A gauntlet, dress rehearsals, kit check

    Train hardest for what weighs most: the on-site cases under competition-identical time limits and device rules.
